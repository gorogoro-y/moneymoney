🛡️ 2アカウント限定 Firestoreセキュリティルール

Firebase Consoleの「Firestore」➔「ルール」タブに以下のコードを貼り付けて「公開」してください。指定の2アカウント以外を完全遮断します。 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAllowedUser() {
      return request.auth != null && 
             request.auth.token.email in ['y37y37@gmail.com', 'nuri0215@gmail.com'];
    }

    match /user_vaults/{userId} {
      allow read, write: if isAllowedUser() && request.auth.uid == userId;
    }

    match /shared_vaults/{sharedDocId} {
      allow read, write: if isAllowedUser();
    }
  }
}
